//@version=6
indicator("Liquidity Structure Engine v5", overlay=true, max_bars_back=1000)

// ═══════════════════════════════════════════════════════════════════════════
//  ARCHITECTURE v5
//  ─────────────────────────────────────────────────────────────────────────
//  Layer 1  │ Session Clock            — Tokyo / London / NY + transitions
//  Layer 2  │ HTF Bias (1H)            — Structure read on higher timeframe
//  Layer 3  │ VWAP (session)           — Institutional mean, session-anchored
//  Layer 4  │ POC Approximation        — Highest volume node in lookback window
//  Layer 5  │ POC/VWAP State           — Balance / Bull Imbalance / Bear Imbalance
//  Layer 6  │ Session-Relative Volume  — Percentile vs same-session bars only
//  Layer 7  │ Swing Structure          — HTF + LTF swing highs/lows
//  Layer 8  │ Regime Detection         — Volatility vs Drift
//  Layer 9  │ Sweep Detection          — Engineered stop runs with quality score
//  Layer 10 │ Sweep Score (1–3)        — Session weight x structure confluence
//  Layer 11 │ Drift Scalp Layer        — Low-vol momentum + candle character
//  Layer 12 │ Tokyo Range Expansion    — Asian range break without volume gate
//  Layer 13 │ Order Block Detection    — Single candle + impulse departure zones
//  Layer 14 │ Consolidation Zones      — Multi-bar range zones
//  Layer 15 │ Structural Exit Logic    — Target-first, backstop-second
//  Layer 16 │ Dashboard HUD            — Sectioned, premium design
//  Layer 17 │ Initial Balance          — First-hour range: Asian / London / NY
// ═══════════════════════════════════════════════════════════════════════════

// ─────────────────────── INPUTS ───────────────────────

htfRes         = input.timeframe("60",  "HTF Resolution (Bias)",               group="Structure")
pocLookback    = input.int(60,          "POC Lookback Bars",                    group="Structure", minval=20, maxval=300)
swingLen       = input.int(20,          "Swing Lookback (LTF)",                 group="Structure", minval=5,  maxval=50)
swingLenHTF    = input.int(40,          "Swing Lookback (HTF)",                 group="Structure", minval=10, maxval=100)

volPercentile  = input.int(75,          "Volume Surge Percentile",              group="Volume",    minval=50, maxval=95)
volLookback    = input.int(80,          "Session Volume Window (bars)",         group="Volume",    minval=30, maxval=200)

sweepWickMin   = input.float(0.7,       "Min Sweep Wick (ATR x)",               group="Sweep",     minval=0.2, maxval=2.0, step=0.05)
pocVwapThresh  = input.float(0.3,       "Balance Threshold (ATR x)",            group="Sweep",     minval=0.1, maxval=0.8, step=0.05)
minSweepScore  = input.int(2,           "Min Score to Signal (1-3)",            group="Sweep",     minval=1,   maxval=3)

regimeLookback = input.int(20,          "Regime ATR Lookback",                  group="Regime",    minval=10, maxval=50)
regimeThresh   = input.float(0.6,       "Drift/Volatile Threshold (ATR ratio)", group="Regime",    minval=0.3, maxval=1.0, step=0.05)
driftBars      = input.int(3,           "Drift Confirmation Bars",              group="Regime",    minval=2,  maxval=6)

driftBodyMin   = input.float(0.4,       "Min Body for Drift Candle (ATR x)",    group="Drift",     minval=0.2, maxval=1.0, step=0.05)
driftVolMin    = input.int(50,          "Min Vol Percentile (Drift)",           group="Drift",     minval=30, maxval=80)
pocLeaveMin    = input.float(0.5,       "POC Leave-Behind (ATR x)",             group="Drift",     minval=0.2, maxval=2.0, step=0.05)

tokyoBreakBody = input.float(0.3,       "Tokyo Break Min Body (ATR x)",         group="Tokyo",     minval=0.1, maxval=1.0, step=0.05)
tokyoConfBars  = input.int(2,           "Tokyo Break Confirmation Bars",        group="Tokyo",     minval=1,  maxval=4)

obImpulseMin   = input.float(1.5,       "OB Impulse Strength (ATR x)",          group="Order Blocks", minval=0.8, maxval=4.0, step=0.1)
obProximity    = input.float(0.3,       "OB Exit Proximity (ATR x)",            group="Order Blocks", minval=0.1, maxval=1.0, step=0.05)

consBars       = input.int(4,           "Consolidation Min Bars",               group="Consolidation", minval=3, maxval=10)
consRangeMax   = input.float(0.5,       "Max Range for Consolidation (ATR x)",  group="Consolidation", minval=0.2, maxval=1.5, step=0.05)

tokyoOpen      = input.int(0,           "Tokyo Open  (UTC hour)",               group="Sessions",  minval=0, maxval=23)
tokyoClose     = input.int(9,           "Tokyo Close (UTC hour)",               group="Sessions",  minval=0, maxval=23)
londonOpen     = input.int(7,           "London Open (UTC hour)",               group="Sessions",  minval=0, maxval=23)
londonClose    = input.int(16,          "London Close(UTC hour)",               group="Sessions",  minval=0, maxval=23)
nyOpen         = input.int(13,          "NY Open     (UTC hour)",               group="Sessions",  minval=0, maxval=23)
nyClose        = input.int(22,          "NY Close    (UTC hour)",               group="Sessions",  minval=0, maxval=23)

atrLen         = input.int(14,          "ATR Length",                           group="Core")

hudPos         = input.string("Top Right", "Panel Position",
     options=["Top Right", "Top Left", "Bottom Right", "Bottom Left"],
     group="Panel")

// ─────────────────────── CORE ───────────────────────

atr  = ta.atr(atrLen)
vwap = ta.vwap(hlc3)

// ─────────────────────── SESSION CLOCK ───────────────────────

currentHour      = hour(time, "UTC")
inTokyo          = currentHour >= tokyoOpen  and currentHour < tokyoClose
inLondon         = currentHour >= londonOpen and currentHour < londonClose
inNY             = currentHour >= nyOpen     and currentHour < nyClose
londonTransition = currentHour == londonOpen
nyTransition     = currentHour == nyOpen

sessionLabel =
     londonTransition ? "LONDON OPEN" :
     nyTransition     ? "NY / COMEX"  :
     inLondon         ? "LONDON"      :
     inNY             ? "NEW YORK"    :
     inTokyo          ? "TOKYO"       :
     "OFF-SESSION"

sessionWeight =
     londonTransition ? 3 :
     nyTransition     ? 3 :
     inLondon         ? 2 :
     inNY             ? 2 :
     inTokyo          ? 1 : 0

// ─────────────────────── HTF BIAS ───────────────────────

htfClose = request.security(syminfo.tickerid, htfRes, close,          lookahead=barmerge.lookahead_off)
htfVwap  = request.security(syminfo.tickerid, htfRes, ta.vwap(hlc3),  lookahead=barmerge.lookahead_off)
htfHigh  = request.security(syminfo.tickerid, htfRes, high,           lookahead=barmerge.lookahead_off)
htfLow   = request.security(syminfo.tickerid, htfRes, low,            lookahead=barmerge.lookahead_off)
htfAtr   = request.security(syminfo.tickerid, htfRes, ta.atr(atrLen), lookahead=barmerge.lookahead_off)

htfBullBias = htfClose > htfVwap
htfBearBias = htfClose < htfVwap
htfNeutral  = math.abs(htfClose - htfVwap) < htfAtr * pocVwapThresh

// ─────────────────────── POC ───────────────────────

var float poc = na
float maxVol  = 0.0
for i = 0 to pocLookback - 1
    if volume[i] > maxVol
        maxVol := volume[i]
        poc    := close[i]

pocVwapDelta   = poc - vwap
pocVwapBalance = math.abs(pocVwapDelta) < atr * pocVwapThresh
bullImbalance  = poc > vwap and not pocVwapBalance
bearImbalance  = poc < vwap and not pocVwapBalance

// ═══════════════════════════════════════════════════════════════════════════
//  LAYER 6 — SESSION-RELATIVE VOLUME NORMALIZATION
// ═══════════════════════════════════════════════════════════════════════════

sessionVolRank  = 0.0
sessionVolCount = 0

for i = 1 to volLookback
    barInTokyo  = hour(time[i], "UTC") >= tokyoOpen  and hour(time[i], "UTC") < tokyoClose
    barInLondon = hour(time[i], "UTC") >= londonOpen and hour(time[i], "UTC") < londonClose
    barInNY     = hour(time[i], "UTC") >= nyOpen     and hour(time[i], "UTC") < nyClose

    sameSession =
         (inTokyo  and barInTokyo)  or
         (inLondon and barInLondon) or
         (inNY     and barInNY)     or
         (not inTokyo and not inLondon and not inNY and
          not barInTokyo and not barInLondon and not barInNY)

    if sameSession
        sessionVolCount += 1
        if volume[i] < volume
            sessionVolRank += 1

volRank = sessionVolCount > 5 ? (sessionVolRank / sessionVolCount * 100) : 50.0

bodySize   = math.abs(close - open)
bodyPct    = bodySize / atr

volSurge   = volRank >= volPercentile
absorption = volSurge and bodyPct < 0.35
bullFlow   = volSurge and close > open and bodyPct > 0.5
bearFlow   = volSurge and close < open and bodyPct > 0.5
driftVolOk = volRank >= driftVolMin

// ─────────────────────── SWING STRUCTURE ───────────────────────

ltfSwingHigh = ta.pivothigh(high,    swingLen,    swingLen)
ltfSwingLow  = ta.pivotlow(low,     swingLen,    swingLen)
htfSwingHigh = ta.pivothigh(htfHigh, swingLenHTF, swingLenHTF)
htfSwingLow  = ta.pivotlow(htfLow,  swingLenHTF, swingLenHTF)

var float lastLTFHigh = na
var float lastLTFLow  = na
var float lastHTFHigh = na
var float lastHTFLow  = na

if not na(ltfSwingHigh)
    lastLTFHigh := ltfSwingHigh
if not na(ltfSwingLow)
    lastLTFLow  := ltfSwingLow
if not na(htfSwingHigh)
    lastHTFHigh := htfSwingHigh
if not na(htfSwingLow)
    lastHTFLow  := htfSwingLow

// ─────────────────────── ASIAN RANGE ───────────────────────

var float asianHigh          = na
var float asianLow           = na
var float asianHighCandidate = na
var float asianLowCandidate  = na

if currentHour == tokyoOpen and currentHour[1] != tokyoOpen
    asianHighCandidate := high
    asianLowCandidate  := low

if inTokyo
    asianHighCandidate := na(asianHighCandidate) ? high : math.max(asianHighCandidate, high)
    asianLowCandidate  := na(asianLowCandidate)  ? low  : math.min(asianLowCandidate,  low)

if currentHour == londonOpen
    asianHigh := asianHighCandidate
    asianLow  := asianLowCandidate

// ═══════════════════════════════════════════════════════════════════════════
//  LAYER 17 — INITIAL BALANCE (ASIAN / LONDON / NY)
//  ─────────────────────────────────────────────────────────────────────────
//  IB = the price range built during the FIRST HOUR of each session open.
//  Source: Market Profile / TPO theory (Peter Steidlmayer).
//
//  Asian IB   — UTC hour == tokyoOpen   (default 00:00 → 01:00)
//  London IB  — UTC hour == londonOpen  (default 07:00 → 08:00)
//  NY IB      — UTC hour == nyOpen      (default 13:00 → 14:00)
//
//  Chart line colors (chosen to not conflict with any existing plots):
//    Asian IB   → violet  rgb(180, 100, 255)
//    London IB  → lime    rgb(130, 255, 100)
//    NY IB      → sky     rgb( 55, 180, 255)
//  Live build = thick + bright; locked = thin + 45% transparent.
//
//  IB break markers (diamond, purple):
//    IB+ = first bar closing above the active session's locked IB high
//    IB- = first bar closing below the active session's locked IB low
//
//  HUD IB STATE logic (row 16):
//    BUILDING   — within the IB hour; range is live, not yet locked
//    ABOVE IB   — close > active IB high  → bullish expansion mode
//    INSIDE IB  — close between H and L   → chop/wait for breakout
//    BELOW IB   — close < active IB low   → bearish expansion mode
//
//  Confluence rules (use with other layers):
//    • IB break + HTF Bias agreement = highest conviction
//    • IB High/Low at an OB zone     = institutional confluence
//    • IB level at Fib 0.50/0.62     = structural magnet / rejection
//    • London IB most important for gold (primary bullion pricing session)
// ═══════════════════════════════════════════════════════════════════════════

inAsianIBHour  = currentHour == tokyoOpen
inLondonIBHour = currentHour == londonOpen
inNYIBHour     = currentHour == nyOpen

// ── Asian IB ──────────────────────────────────────────────────────────────
var float asianIBHighBuild = na
var float asianIBLowBuild  = na
var float asianIBHigh      = na
var float asianIBLow       = na

// Reset build state at the very first bar of the IB hour each day
if inAsianIBHour and not inAsianIBHour[1]
    asianIBHighBuild := high
    asianIBLowBuild  := low

// Expand range while inside the IB hour
if inAsianIBHour
    asianIBHighBuild := na(asianIBHighBuild) ? high : math.max(asianIBHighBuild, high)
    asianIBLowBuild  := na(asianIBLowBuild)  ? low  : math.min(asianIBLowBuild,  low)

// Lock on the first bar after the IB hour ends
if not inAsianIBHour and inAsianIBHour[1]
    asianIBHigh := asianIBHighBuild
    asianIBLow  := asianIBLowBuild

// ── London IB ─────────────────────────────────────────────────────────────
var float londonIBHighBuild = na
var float londonIBLowBuild  = na
var float londonIBHigh      = na
var float londonIBLow       = na

if inLondonIBHour and not inLondonIBHour[1]
    londonIBHighBuild := high
    londonIBLowBuild  := low

if inLondonIBHour
    londonIBHighBuild := na(londonIBHighBuild) ? high : math.max(londonIBHighBuild, high)
    londonIBLowBuild  := na(londonIBLowBuild)  ? low  : math.min(londonIBLowBuild,  low)

if not inLondonIBHour and inLondonIBHour[1]
    londonIBHigh := londonIBHighBuild
    londonIBLow  := londonIBLowBuild

// ── NY IB ──────────────────────────────────────────────────────────────────
var float nyIBHighBuild = na
var float nyIBLowBuild  = na
var float nyIBHigh      = na
var float nyIBLow       = na

if inNYIBHour and not inNYIBHour[1]
    nyIBHighBuild := high
    nyIBLowBuild  := low

if inNYIBHour
    nyIBHighBuild := na(nyIBHighBuild) ? high : math.max(nyIBHighBuild, high)
    nyIBLowBuild  := na(nyIBLowBuild)  ? low  : math.min(nyIBLowBuild,  low)

if not inNYIBHour and inNYIBHour[1]
    nyIBHigh := nyIBHighBuild
    nyIBLow  := nyIBLowBuild

// ── Active IB Reference ────────────────────────────────────────────────────
// During the IB building hour → use live candidates so HUD is live.
// After hour locks → use the locked level.
// Off-session / no data → na (HUD shows "--").
activeIBHigh =
     inAsianIBHour  ? asianIBHighBuild  :
     inLondonIBHour ? londonIBHighBuild :
     inNYIBHour     ? nyIBHighBuild     :
     inTokyo        ? asianIBHigh       :
     inLondon       ? londonIBHigh      :
     inNY           ? nyIBHigh          : na

activeIBLow =
     inAsianIBHour  ? asianIBLowBuild   :
     inLondonIBHour ? londonIBLowBuild  :
     inNYIBHour     ? nyIBLowBuild      :
     inTokyo        ? asianIBLow        :
     inLondon       ? londonIBLow       :
     inNY           ? nyIBLow           : na

ibBuilding = inAsianIBHour or inLondonIBHour or inNYIBHour
ibAbove    = not na(activeIBHigh) and close > activeIBHigh and not ibBuilding
ibBelow    = not na(activeIBLow)  and close < activeIBLow  and not ibBuilding
ibInside   = not ibAbove and not ibBelow and not na(activeIBHigh) and not ibBuilding

// First-bar break detection (for IB+ / IB- chart markers)
ibBullBreak = ibAbove and not ibAbove[1]
ibBearBreak = ibBelow and not ibBelow[1]

// ─────────────────────── REGIME DETECTION ───────────────────────

atrMean        = ta.sma(atr, regimeLookback)
atrRatio       = atr / atrMean
regimeVolatile = atrRatio > (1.0 + regimeThresh)
regimeDrift    = atrRatio < (1.0 - regimeThresh * 0.5)
regimeNeutral  = not regimeVolatile and not regimeDrift

regimeLabel =
     regimeVolatile ? "VOLATILE" :
     regimeDrift    ? "DRIFT"    :
     "NEUTRAL"

// ─────────────────────── CANDLE CHARACTER ───────────────────────

bullCandle = close > open and bodySize > atr * driftBodyMin
bearCandle = close < open and bodySize > atr * driftBodyMin

var int bullCount = 0
var int bearCount = 0
bullCount := bullCandle ? bullCount + 1 : 0
bearCount := bearCandle ? bearCount + 1 : 0

bullMomentum = bullCount >= driftBars
bearMomentum = bearCount >= driftBars

// ─────────────────────── MICRO STRUCTURE ───────────────────────

bullMicroStr = high > high[1] and low > low[1]
bearMicroStr = high < high[1] and low < low[1]

var int bullMicroCount = 0
var int bearMicroCount = 0
bullMicroCount := bullMicroStr ? bullMicroCount + 1 : 0
bearMicroCount := bearMicroStr ? bearMicroCount + 1 : 0

bullMicroConfirmed = bullMicroCount >= driftBars
bearMicroConfirmed = bearMicroCount >= driftBars

// ─────────────────────── POC LEAVE-BEHIND ───────────────────────

pocLeftBehindBull = close > poc and (close - poc) > atr * pocLeaveMin
pocLeftBehindBear = close < poc and (poc - close) > atr * pocLeaveMin

// ─────────────────────── VWAP RIDE ───────────────────────

var int barsAboveVwap = 0
var int barsBelowVwap = 0
barsAboveVwap := close > vwap ? barsAboveVwap + 1 : 0
barsBelowVwap := close < vwap ? barsBelowVwap + 1 : 0

vwapRideBull = barsAboveVwap >= driftBars
vwapRideBear = barsBelowVwap >= driftBars

// ─────────────────────── DRIFT CONDITIONS ───────────────────────

driftLong =
     regimeDrift        and
     bullMomentum       and
     bullMicroConfirmed and
     pocLeftBehindBull  and
     vwapRideBull       and
     htfBullBias        and
     driftVolOk

driftShort =
     regimeDrift        and
     bearMomentum       and
     bearMicroConfirmed and
     pocLeftBehindBear  and
     vwapRideBear       and
     htfBearBias        and
     driftVolOk

// ─────────────────────── LAYER 12: TOKYO RANGE EXPANSION ───────────────────────

var int tokyoBars = 0
tokyoBars := inTokyo ? tokyoBars + 1 : 0

tokyoRangeReady = tokyoBars >= 10 and not na(asianHighCandidate) and not na(asianLowCandidate)

liveAsianHigh = na(asianHighCandidate) ? asianHigh : asianHighCandidate
liveAsianLow  = na(asianLowCandidate)  ? asianLow  : asianLowCandidate

tokyoBullBreak = inTokyo and tokyoRangeReady and
     not na(liveAsianHigh) and
     close > liveAsianHigh and
     bodySize > atr * tokyoBreakBody and
     close > open

tokyoBearBreak = inTokyo and tokyoRangeReady and
     not na(liveAsianLow) and
     close < liveAsianLow and
     bodySize > atr * tokyoBreakBody and
     close < open

var int tokyoBullBreakCount = 0
var int tokyoBearBreakCount = 0
tokyoBullBreakCount := tokyoBullBreak ? tokyoBullBreakCount + 1 : 0
tokyoBearBreakCount := tokyoBearBreak ? tokyoBearBreakCount + 1 : 0

tokyoLong  = tokyoBullBreakCount == tokyoConfBars and htfBullBias
tokyoShort = tokyoBearBreakCount == tokyoConfBars and htfBearBias

// ─────────────────────── SWEEP DETECTION ───────────────────────

bullSweepLTF   = false
bullSweepHTF   = false
bullSweepAsian = false
bearSweepLTF   = false
bearSweepHTF   = false
bearSweepAsian = false

if not na(lastLTFLow)
    bullSweepLTF := low < lastLTFLow and close > lastLTFLow and (lastLTFLow - low) > atr * sweepWickMin
if not na(lastHTFLow)
    bullSweepHTF := low < lastHTFLow and close > lastHTFLow and (lastHTFLow - low) > atr * sweepWickMin
if not na(asianLow) and londonTransition
    bullSweepAsian := low < asianLow and close > asianLow and (asianLow - low) > atr * sweepWickMin * 0.7

if not na(lastLTFHigh)
    bearSweepLTF := high > lastLTFHigh and close < lastLTFHigh and (high - lastLTFHigh) > atr * sweepWickMin
if not na(lastHTFHigh)
    bearSweepHTF := high > lastHTFHigh and close < lastHTFHigh and (high - lastHTFHigh) > atr * sweepWickMin
if not na(asianHigh) and londonTransition
    bearSweepAsian := high > asianHigh and close < asianHigh and (high - asianHigh) > atr * sweepWickMin * 0.7

bullSweep = bullSweepLTF or bullSweepHTF or bullSweepAsian
bearSweep = bearSweepLTF or bearSweepHTF or bearSweepAsian

// ─────────────────────── SWEEP SCORING ───────────────────────

bullSweepScore = 0
bearSweepScore = 0

if bullSweep
    bullSweepScore := 1
    if bullImbalance or (poc > vwap and not pocVwapBalance)
        bullSweepScore += 1
    if sessionWeight >= 2
        bullSweepScore += 1

if bearSweep
    bearSweepScore := 1
    if bearImbalance or (poc < vwap and not pocVwapBalance)
        bearSweepScore += 1
    if sessionWeight >= 2
        bearSweepScore += 1

bullSweepScore := htfBearBias and not htfNeutral ? math.min(bullSweepScore, 1) : bullSweepScore
bearSweepScore := htfBullBias and not htfNeutral ? math.min(bearSweepScore, 1) : bearSweepScore

// ─────────────────────── FAKEOUT ───────────────────────

var bool prevBullBreakout = false
var bool prevBearBreakout = false

bullBreakout = not na(lastLTFLow)  and close < lastLTFLow  and volSurge
bearBreakout = not na(lastLTFHigh) and close > lastLTFHigh and volSurge

prevBullBreakout := bullBreakout[1]
prevBearBreakout := bearBreakout[1]

bullFakeout = prevBullBreakout and close > lastLTFLow[1]
bearFakeout = prevBearBreakout and close < lastLTFHigh[1]

// ─────────────────────── ORDER BLOCKS ───────────────────────

bullImpulse = (close - open) > atr * obImpulseMin and close > open
bearImpulse = (open - close) > atr * obImpulseMin and close < open

var float bearOBHigh  = na
var float bearOBLow   = na
var float bullOBHigh  = na
var float bullOBLow   = na
var bool  bearOBValid = false
var bool  bullOBValid = false

if bearImpulse and close[1] > open[1]
    bearOBHigh  := high[1]
    bearOBLow   := low[1]
    bearOBValid := true

if bullImpulse and close[1] < open[1]
    bullOBHigh  := high[1]
    bullOBLow   := low[1]
    bullOBValid := true

if bearOBValid and not na(bearOBLow) and close > bearOBHigh
    bearOBValid := false
if bullOBValid and not na(bullOBHigh) and close < bullOBLow
    bullOBValid := false

nearBearOB = bearOBValid and not na(bearOBHigh) and high >= (bearOBLow - atr * obProximity)
nearBullOB = bullOBValid and not na(bullOBLow)  and low  <= (bullOBHigh + atr * obProximity)

// ─────────────────────── CONSOLIDATION ZONES ───────────────────────

consRangeHigh = ta.highest(high, consBars)
consRangeLow  = ta.lowest(low,   consBars)
consRange     = consRangeHigh - consRangeLow
isConsolidating = consRange < atr * consRangeMax

var float consZoneHigh   = na
var float consZoneLow    = na
var bool  consZoneValid  = false
var bool  wasConsolidating = false

if wasConsolidating and not isConsolidating
    consZoneHigh  := consRangeHigh[1]
    consZoneLow   := consRangeLow[1]
    consZoneValid := true

wasConsolidating := isConsolidating

if consZoneValid
    if close > consZoneHigh + atr * 0.2 or close < consZoneLow - atr * 0.2
        consZoneValid := false

reEntryFromBelow = consZoneValid and not na(consZoneLow) and
     close[1] < consZoneLow and close >= consZoneLow

reEntryFromAbove = consZoneValid and not na(consZoneHigh) and
     close[1] > consZoneHigh and close <= consZoneHigh

// ─────────────────────── ENTRY CONDITIONS ───────────────────────

longCondition =
     (regimeVolatile and bullSweep and bullSweepScore >= minSweepScore and (bullFlow or absorption)) or
     (regimeNeutral  and bullSweep and bullSweepScore == 3             and (bullFlow or absorption)) or
     driftLong or
     tokyoLong

shortCondition =
     (regimeVolatile and bearSweep and bearSweepScore >= minSweepScore and (bearFlow or absorption)) or
     (regimeNeutral  and bearSweep and bearSweepScore == 3             and (bearFlow or absorption)) or
     driftShort or
     tokyoShort

// ─────────────────────── STRUCTURAL EXIT TARGETS ───────────────────────

f_longTarget(entryPrice) =>
    target = float(na)
    if bearOBValid and not na(bearOBLow) and bearOBLow > entryPrice
        if na(target) or bearOBLow < target
            target := bearOBLow
    if consZoneValid and not na(consZoneHigh) and consZoneLow > entryPrice
        if na(target) or consZoneLow < target
            target := consZoneLow
    if not na(lastLTFHigh) and lastLTFHigh > entryPrice
        if na(target) or lastLTFHigh < target
            target := lastLTFHigh
    if not na(lastHTFHigh) and lastHTFHigh > entryPrice
        if na(target) or lastHTFHigh < target
            target := lastHTFHigh
    if not na(poc) and poc > entryPrice
        if na(target) or poc < target
            target := poc
    target

f_shortTarget(entryPrice) =>
    target = float(na)
    if bullOBValid and not na(bullOBHigh) and bullOBHigh < entryPrice
        if na(target) or bullOBHigh > target
            target := bullOBHigh
    if consZoneValid and not na(consZoneLow) and consZoneHigh < entryPrice
        if na(target) or consZoneHigh > target
            target := consZoneHigh
    if not na(lastLTFLow) and lastLTFLow < entryPrice
        if na(target) or lastLTFLow > target
            target := lastLTFLow
    if not na(lastHTFLow) and lastHTFLow < entryPrice
        if na(target) or lastHTFLow > target
            target := lastHTFLow
    if not na(poc) and poc < entryPrice
        if na(target) or poc > target
            target := poc
    target

var int   pos              = 0
var float longEntryPrice   = na
var float shortEntryPrice  = na
var float longTargetLevel  = na
var float shortTargetLevel = na
var bool  inDriftLong      = false
var bool  inDriftShort     = false
var bool  inTokyoLong      = false
var bool  inTokyoShort     = false

buySignal  = false
sellSignal = false
exitLong   = false
exitShort  = false

if pos == 0
    if longCondition
        buySignal       := true
        pos             := 1
        inDriftLong     := driftLong
        inTokyoLong     := tokyoLong
        inDriftShort    := false
        inTokyoShort    := false
        longEntryPrice  := close
        longTargetLevel := f_longTarget(close)

    else if shortCondition
        sellSignal       := true
        pos              := -1
        inDriftShort     := driftShort
        inTokyoShort     := tokyoShort
        inDriftLong      := false
        inTokyoLong      := false
        shortEntryPrice  := close
        shortTargetLevel := f_shortTarget(close)

else if pos == 1
    primaryExitLong  = not na(longTargetLevel) and high >= longTargetLevel
    consExitLong     = reEntryFromBelow
    obExitLong       = nearBearOB
    tokyoExitLong    = inTokyoLong and (not inTokyo or
         (not na(liveAsianLow) and low <= liveAsianLow))

    backstopLong =
         (bearSweep and bearSweepScore >= 2) or
         (close < vwap and bearFlow and not inDriftLong and not inTokyoLong)

    if primaryExitLong or consExitLong or obExitLong or tokyoExitLong
        exitLong        := true
        pos             := 0
        longTargetLevel := na
        inDriftLong     := false
        inTokyoLong     := false

    else if backstopLong
        exitLong        := true
        pos             := 0
        longTargetLevel := na
        inDriftLong     := false
        inTokyoLong     := false

    if pos == 1 and shortCondition
        exitLong         := true
        sellSignal       := true
        pos              := -1
        inDriftShort     := driftShort
        inTokyoShort     := tokyoShort
        inDriftLong      := false
        inTokyoLong      := false
        longTargetLevel  := na
        shortEntryPrice  := close
        shortTargetLevel := f_shortTarget(close)

else if pos == -1
    primaryExitShort = not na(shortTargetLevel) and low <= shortTargetLevel
    consExitShort    = reEntryFromAbove
    obExitShort      = nearBullOB
    tokyoExitShort   = inTokyoShort and (not inTokyo or
         (not na(liveAsianHigh) and high >= liveAsianHigh))

    backstopShort =
         (bullSweep and bullSweepScore >= 2) or
         (close > vwap and bullFlow and not inDriftShort and not inTokyoShort)

    if primaryExitShort or consExitShort or obExitShort or tokyoExitShort
        exitShort        := true
        pos              := 0
        shortTargetLevel := na
        inDriftShort     := false
        inTokyoShort     := false

    else if backstopShort
        exitShort        := true
        pos              := 0
        shortTargetLevel := na
        inDriftShort     := false
        inTokyoShort     := false

    if pos == -1 and longCondition
        exitShort        := true
        buySignal        := true
        pos              := 1
        inDriftLong      := driftLong
        inTokyoLong      := tokyoLong
        inDriftShort     := false
        inTokyoShort     := false
        shortTargetLevel := na
        longEntryPrice   := close
        longTargetLevel  := f_longTarget(close)
